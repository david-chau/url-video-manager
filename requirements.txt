# yt-dlp is deliberately unpinned: YouTube breaks extractors on the order of
# monthly, and a pinned version silently rots into "everything fails". The
# Dockerfile entrypoint runs `pip install -U 'yt-dlp[default]'` before every
# start, so `docker compose restart` is always the first fix to try.
# [default] pulls in yt-dlp-ejs, the external JS runtime bridge YouTube's
# "n challenge" now requires (see Dockerfile's Deno install) -- bare yt-dlp
# fails every download with "Requested format is not available" once
# YouTube's challenge kicks in, even with valid cookies.
yt-dlp[default]

pysrt
